  <script>
    // 1. STRICT SECURITY GATE:
    // Forces re-login if the tab was closed or opened fresh without the active session token.
    if (!sessionStorage.getItem('sgi_admin_active')) {
      window.location.replace('/admin/logout');
    }

    // Handles browser Back button from cache
    window.addEventListener('pageshow', (event) => {
      if (event.persisted || !sessionStorage.getItem('sgi_admin_active')) {
        window.location.replace('/admin/logout');
      }
    });

    // 2. DEDICATED LOGOUT FUNCTION
    function handleLogout() {
      sessionStorage.removeItem('sgi_admin_active');
      window.location.replace('/admin/logout');
    }

    function toggleProfileDrawer() {
      const drawer = document.getElementById('profileDrawer');
      drawer.classList.toggle('open');
    }

    // Close drawer when clicking outside
    window.addEventListener('click', (e) => {
      const drawer = document.getElementById('profileDrawer');
      const profileSec = document.querySelector('.admin-profile-section');
      if (drawer && !drawer.contains(e.target) && !profileSec.contains(e.target)) {
        drawer.classList.remove('open');
      }
    });

    let activeBranch = 'all';

    function setBranchFilter(branch, btnElem) {
      activeBranch = branch;
      document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
      if (btnElem) btnElem.classList.add('active');
      document.getElementById('exportBtn').textContent = `Download ${branch === 'all' ? 'All' : branch} Excel 📥`;
      loadLeads();
    }

    function downloadBranchExcel() {
      window.location.href = `/api/export-leads?branch=${encodeURIComponent(activeBranch)}`;
    }

    async function loadLeads() {
      try {
        const url = activeBranch === 'all' ? '/api/leads' : `/api/leads?branch=${encodeURIComponent(activeBranch)}`;
        const res = await fetch(url);

        if (res.status === 401) {
          handleLogout();
          return;
        }

        const data = await res.json();
        const tbody = document.getElementById('leadsTableBody');

        if (!data.leads || data.leads.length === 0) {
          tbody.innerHTML = `<tr><td colspan="7" class="empty-state">No inquiries recorded for ${activeBranch === 'all' ? 'any branch' : activeBranch} yet.</td></tr>`;
          return;
        }

        tbody.innerHTML = data.leads.map(lead => {
          const cleanPhone = lead.phone.replace(/[^0-9]/g, '');
          const waPhone = cleanPhone.length === 10 ? `91${cleanPhone}` : cleanPhone;
          const waMsg = encodeURIComponent(`Hello ${lead.name}, thank you for inquiring about admission in ${lead.program} at Sukhjinder Group of Institutes (SGI), Gurdaspur.`);
          return `
            <tr>
              <td>#${lead.id}</td>
              <td style="font-weight:700;">${lead.name}</td>
              <td>${lead.phone}</td>
              <td>${lead.program}</td>
              <td><span class="branch-pill">${lead.branch}</span></td>
              <td style="color:var(--text-muted); font-size:0.82rem;">${lead.timestamp}</td>
              <td>
                <div class="action-links">
                  <a href="https://wa.me/${waPhone}?text=${waMsg}" target="_blank" class="btn-act btn-wa">WhatsApp 💬</a>
                  <a href="tel:${cleanPhone}" class="btn-act btn-call">Call 📞</a>
                </div>
              </td>
            </tr>
          `;
        }).join('');
      } catch (err) {
        console.error("Sync error:", err);
      }
    }

    async function loadStats() {
      try {
        const res = await fetch('/api/admin/stats');
        if (res.status === 401) {
          handleLogout();
          return;
        }
        const data = await res.json();
        document.getElementById('statTotal').textContent = data.total;
        document.getElementById('statEngg').textContent = data.engineering;
        document.getElementById('statPharma').textContent = data.pharmacy;
        document.getElementById('statNursing').textContent = data.nursing;
        document.getElementById('statMgmt').textContent = data.management;
      } catch (err) {
        console.error("Stats error:", err);
      }
    }

    async function loadAdminProfile() {
      try {
        const res = await fetch('/api/admin/profile');
        if (res.ok) {
          const data = await res.json();
          document.getElementById('profileName').textContent = data.name || "Director of Admissions";
          document.getElementById('profileRole').textContent = data.role || "Super Admin • SGI Portal";
          document.getElementById('drawerDesignation').textContent = data.name || "Director of Admissions";
          document.getElementById('drawerSessionTime').textContent = data.session_time || "Active Now";
        }
      } catch (e) {}
    }

    // Initial load
    loadAdminProfile();
    loadStats();
    loadLeads();

    // Auto-poll every 4 seconds without logging the user out during active sessions
    setInterval(() => {
      loadStats();
      loadLeads();
    }, 4000);
  </script>
